# PowerSMP -- LuckPerms setup
# Paste these into console in order. Generated from kits.yml assignments.
#
# Creates a 'powersmp' group carrying the kit permission plus the NoCheatPlus
# exemptions the kits need, then puts all 13 players in it.

lp creategroup powersmp

# The kit itself.
lp group powersmp permission set powersmp.use true

# NoCheatPlus. Every kit moves players by setting velocity directly, which is
# exactly what NCP blocks. Without these, abilities get cancelled and players
# get flagged and kicked mid-fight.
lp group powersmp permission set nocheatplus.checks.moving true
lp group powersmp permission set nocheatplus.checks.moving.survivalfly true
lp group powersmp permission set nocheatplus.checks.moving.creativefly true
lp group powersmp permission set nocheatplus.checks.moving.morepackets true
lp group powersmp permission set nocheatplus.checks.moving.nofall true
lp group powersmp permission set nocheatplus.checks.moving.passable true
# Reach: Mavricc's blue stance and consolidation extend reach through
# attributes, which NCP does not read, so it reads as reach hacking.
lp group powersmp permission set nocheatplus.checks.fight.reach true
# Attack speed: ItzMeTentx in water, Mavricc red+affinity, Made In Heaven.
lp group powersmp permission set nocheatplus.checks.fight.speed true
# Fast block breaking: Marb13_'s Haste V and Mavricc's blue stance.
lp group powersmp permission set nocheatplus.checks.blockbreak.fastbreak true
